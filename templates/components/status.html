{% macro status_badge(status) %} {%- set s = (status or 'unknown')|lower -%} {%- if s in ['completed', 'validated', 'approved', 'active', 'enabled', 'connected'] -%} {%- set tone = 'border-success/25 bg-success/10 text-success' -%} {%- elif s in ['failed', 'error', 'cancelled', 'forbidden', 'disabled'] -%} {%- set tone = 'border-error/25 bg-error/10 text-error' -%} {%- elif s in ['queued', 'preparing', 'parsing', 'segmenting', 'translating', 'finalizing', 'cancelling', 'draft', 'pending', 'running', 'in_progress'] -%} {%- set tone = 'border-info/25 bg-info/10 text-info' -%} {%- else -%} {%- set tone = 'border-base-300 bg-base-200 text-base-content/60' -%} {%- endif -%} {{ (status or 'unknown')|replace('_', ' ') }} {% endmacro %}
